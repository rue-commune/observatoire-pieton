{{- /* Returns the city's config JSON, verbatim except for the tile URL. The JSON is passed through untouched rather than re-serialised, because jsonify would sort keys and the app relies on their order (e.g. the marking types). Only the "tuiles" value is substituted: * params.tilesurl (HUGO_PARAMS_TILESURL) replaces it outright, e.g. with the same-origin tile proxy in production; * otherwise, outside the production environment only (hugo server, or --environment development), params.cartokey (HUGO_PARAMS_CARTOKEY, from the git-ignored .env) is appended as the key query parameter. Production builds ignore it, so no production artefact can carry the key, whatever the environment. With neither applying, the CARTO URL is used without a key. */ -}} {{- $json := strings.TrimSuffix "\n" .json -}} {{- $default := .cfg.tuiles -}} {{- $tiles := $default -}} {{- with site.Params.tilesurl -}} {{- $tiles = . -}} {{- else -}} {{- if not hugo.IsProduction -}} {{- with site.Params.cartokey -}} {{- $sep := cond (strings.Contains $default "?") "&" "?" -}} {{- $tiles = printf "%s%skey=%s" $default $sep . -}} {{- end -}} {{- end -}} {{- end -}} {{- if ne $tiles $default -}} {{- $old := printf "\"tuiles\": %s" (jsonify $default) -}} {{- if not (strings.Contains $json $old) -}} {{- errorf "config-tuiles: %s not found verbatim in city config" $old -}} {{- end -}} {{- $json = replace $json $old (printf "\"tuiles\": %s" (jsonify $tiles)) 1 -}} {{- end -}} {{- return $json -}}
