{{- /* Figures from the page's city data, computed at build time so the text follows each data update. Usage: {{< donnees "places" >}} spaces to remove to comply with the law (upstream side), as on the map's "places à supprimer" panel, rounded down: "plus de 3 000". {{< donnees "date" >}} the data date: "5 novembre 2026". */ -}} {{- $v := partial "ville.html" .Page -}} {{- $quoi := .Get 0 -}} {{- if eq $quoi "places" -}} {{- /* Same categories as drawPlaces() in observatoire.js: cars, loading bays, motorbikes and disabled-badge spaces; index 0 is the upstream (legal) count. */ -}} {{- $n := 0 -}} {{- $vus := 0 -}} {{- range $v.data.removal.c -}} {{- range $cat, $c := . -}} {{- if in (slice "1" "2" "4" "8") $cat -}} {{- $n = add $n (index $c 0) -}} {{- $vus = add $vus 1 -}} {{- end -}} {{- end -}} {{- end -}} {{- /* A total of 0 is legitimate once every crossing complies, but finding no category at all means data.json changed shape: fail rather than publish "0 places". */ -}} {{- if not $vus -}} {{- errorf "%s: data.json removal.c has no category 1, 2, 4 or 8; has its format changed?" .Page.File.Path -}} {{- end -}} {{- /* Round down to the hundred (the ten below 1 000; exact below 100) and say "plus de" only when rounding lost something. */ -}} {{- $pas := cond (ge $n 1000) 100 (cond (ge $n 100) 10 1) -}} {{- $arrondi := mul (div (int $n) $pas) $pas -}} {{- $fmt := lang.FormatNumberCustom 0 $arrondi "-|,|\u00a0" "|" -}} {{- if lt $arrondi $n -}}plus de {{ $fmt }}{{- else -}}{{ $fmt }}{{- end -}} {{- else if eq $quoi "date" -}} {{- $t := time.AsTime $v.data.generated -}} {{- /* French ordinal for the first of the month: "1er novembre". */ -}} {{- if eq $t.Day 1 -}}1er {{ time.Format "January 2006" $t }}{{- else -}}{{ time.Format ":date_long" $t }}{{- end -}} {{- else -}} {{- errorf "%s: donnees shortcode takes \"places\" or \"date\", not %q" .Page.File.Path $quoi -}} {{- end -}}
